cmake_minimum_required(VERSION 3.22)
project(DistillazioneEQ LANGUAGES C CXX)
include(CTest)

add_library(eq_response src/AnalogResponse.cpp src/RecoveredParameters.cpp src/RootSection.cpp src/DigitalSection.cpp src/FilterPrototype.cpp src/StaticEQ.cpp)
set_target_properties(eq_response PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(eq_response PUBLIC src)
target_compile_features(eq_response PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(eq_response PRIVATE /W4)
  add_compile_definitions(_USE_MATH_DEFINES)
else()
  target_compile_options(eq_response PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(BUILD_PLUGIN "Build the audio plug-in and standalone application" ON)
option(AESTER_BUILD_AAX "Build AAX Native development binaries (PACE signing required for release)" OFF)
set(AESTER_AAX_SDK_PATH "" CACHE PATH "Optional external Avid AAX SDK; empty uses bundled JUCE SDK")
if(BUILD_PLUGIN)
  set(JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE" CACHE PATH "Local JUCE source")
  add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE" EXCLUDE_FROM_ALL)
  if(WIN32)
    set(AESTER_FORMATS VST3 Standalone)
  else()
    set(AESTER_FORMATS AU Standalone)
  endif()
  if(AESTER_BUILD_AAX)
    if(AESTER_AAX_SDK_PATH)
      juce_set_aax_sdk_path("${AESTER_AAX_SDK_PATH}")
    endif()
    list(APPEND AESTER_FORMATS AAX)
  endif()
  juce_add_plugin(DistillazioneEQ
    VERSION 1.2.0 COMPANY_NAME "Neurora" COMPANY_WEBSITE "https://www.neurora.it" COMPANY_COPYRIGHT "Copyright (c) 2026 Neurora"
    BUNDLE_ID "it.neurora.aestereq"
    PLUGIN_MANUFACTURER_CODE NrAu PLUGIN_CODE AeEQ
    FORMATS ${AESTER_FORMATS} PRODUCT_NAME "Aester EQ"
    AAX_IDENTIFIER "it.neurora.aestereq.aax" AAX_CATEGORY EQ
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(DistillazioneEQ PRIVATE plugin/Processor.cpp plugin/Editor.cpp)
  target_include_directories(DistillazioneEQ PRIVATE plugin src)
  target_compile_features(DistillazioneEQ PUBLIC cxx_std_17)
  target_compile_definitions(DistillazioneEQ PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(DistillazioneEQ PRIVATE eq_response juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
  if(BUILD_TESTING)
    add_executable(processor_tests tests/ProcessorTests.cpp)
    target_include_directories(processor_tests PRIVATE "$<TARGET_PROPERTY:DistillazioneEQ,INCLUDE_DIRECTORIES>" plugin src)
    target_compile_definitions(processor_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(processor_tests PRIVATE DistillazioneEQ eq_response)
    if(MSVC)
      # The test fixture holds multiple processors on the stack.
      target_link_options(processor_tests PRIVATE /STACK:16777216)
    endif()
    add_executable(characterize tools/Characterize.cpp)
    target_include_directories(characterize PRIVATE "$<TARGET_PROPERTY:DistillazioneEQ,INCLUDE_DIRECTORIES>" plugin src)
    target_compile_definitions(characterize PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(characterize PRIVATE DistillazioneEQ eq_response)
    add_test(NAME plugin_processor COMMAND processor_tests)
  endif()
endif()

if(BUILD_TESTING)
  add_executable(response_tests tests/AnalogResponseTests.cpp)
  target_link_libraries(response_tests PRIVATE eq_response)
  add_test(NAME response_math COMMAND response_tests)
  add_executable(parameter_tests tests/RecoveredParametersTests.cpp)
  target_link_libraries(parameter_tests PRIVATE eq_response)
  add_test(NAME recovered_parameters COMMAND parameter_tests)
  add_executable(root_section_tests tests/RootSectionTests.cpp)
  target_link_libraries(root_section_tests PRIVATE eq_response)
  add_test(NAME root_sections COMMAND root_section_tests)
  add_executable(digital_section_tests tests/DigitalSectionTests.cpp)
  target_link_libraries(digital_section_tests PRIVATE eq_response)
  add_test(NAME digital_sections COMMAND digital_section_tests)
  add_executable(prototype_tests tests/FilterPrototypeTests.cpp)
  target_link_libraries(prototype_tests PRIVATE eq_response)
  add_test(NAME filter_prototypes COMMAND prototype_tests)
  add_executable(static_eq_tests tests/StaticEQTests.cpp)
  target_link_libraries(static_eq_tests PRIVATE eq_response)
  add_test(NAME static_eq COMMAND static_eq_tests)
endif()
